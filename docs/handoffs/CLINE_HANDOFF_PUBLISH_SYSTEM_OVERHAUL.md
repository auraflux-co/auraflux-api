## Context

The publish system has several disconnected pieces that need to be wired together into a single coherent flow. Right now:

- `thumbnail_url` and `first_comment` are conditionally sent — often silently dropped
- Hashtags generated by Claude in publish-copy never reach Upload-Post
- Tags only sent if frontend passes them — publish-copy output is disconnected from publish call
- `categoryId` hardcoded to `"24"` regardless of content type
- TikTok caption truncated to 90 chars — wrong, limit is 2200 runes
- Chapter timestamps in description — YouTube doesn't always auto-parse
- No post-publish outcome card in dashboard
- No stored title alternatives for post-publish switching

**Reference specs:**
- `docs/specs/PUBLISH_COPY_SPEC.md` — authoritative format for all publish copy
- `docs/architecture/PIPELINE_CONTRACT_SPEC.md` — Gate 4 publish package audit

### Timestamp generation

Timestamps must be derived from actual segment durations — not estimated. The endpoint already receives `chapters` as a param. If `segments[]` with `durationSeconds` are available, generate timestamps from running total. If only `chapters` string is passed, use it verbatim.

Running total logic:
```javascript
let runningSeconds = 0;
const lines = ['0:00 Intro'];
for (const seg of segments) {
  runningSeconds += seg.durationSeconds;
  const mm = Math.floor(runningSeconds / 60);
  const ss = String(runningSeconds % 60).padStart(2, '0');
  lines.push(`${mm}:${ss} ${formatSegmentLabel(seg.label, contentType)}`);
}
const chaptersBlock = lines.join('\n');
```

### JavaScript functions to add:

```javascript
function tryAnotherTitle(jobId) {
  const job = JOBS.find(j => j.id === jobId);
  if (!job?.publishCopy?.youtube?.titles) return;
  
  const panel = document.getElementById(`pp-titles-${jobId}`);
  const optionsEl = document.getElementById(`pp-title-options-${jobId}`);
  
  // Render stored title options (excluding current recommended)
  const titles = job.publishCopy.youtube.titles;
  const current = job.publishCopy.youtube.recommended || 0;
  
  optionsEl.innerHTML = titles
    .map((t, i) => i === current ? '' : `
      <label class="pp-title-option">
        <input type="radio" name="title-${jobId}" value="${i}">
        <span>${t}</span>
      </label>
    `).join('');
  
  panel.style.display = 'block';
}

async function applySelectedTitle(jobId) {
  const selected = document.querySelector(`input[name="title-${jobId}"]:checked`);
  if (!selected) return;
  
  const job = JOBS.find(j => j.id === jobId);
  const newTitle = job.publishCopy.youtube.titles[parseInt(selected.value)];
  
  // POST to server to update YouTube title via YouTube API
  const resp = await fetch(`/job/${jobId}/update-title`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ title: newTitle })
  });
  
  if (resp.ok) {
    document.querySelector(`#post-publish-${jobId} .pp-title-current`).textContent = newTitle;
    document.getElementById(`pp-titles-${jobId}`).style.display = 'none';
    logToJob(jobId, `Title updated to: ${newTitle}`);
  }
}

function showPostPublishCard(jobId) {
  const job = JOBS.find(j => j.id === jobId);
  if (!job) return;
  
  const card = document.getElementById(`post-publish-${jobId}`);
  if (!card) return;
  
  // Calculate end screen timestamp (video duration - 20 seconds)
  const duration = job.totalDuration || 0;
  const endScreenSecs = Math.max(0, duration - 20);
  const mm = Math.floor(endScreenSecs / 60);
  const ss = String(endScreenSecs % 60).padStart(2, '0');
  
  // Populate template vars
  card.innerHTML = card.innerHTML
    .replace(/{{jobId}}/g, jobId)
    .replace(/{{title}}/g, job.publishCopy?.youtube?.title || job.title || '')
    .replace(/{{endScreenTimestamp}}/g, `${mm}:${ss}`)
    .replace(/{{studioUrl}}/g, `https://studio.youtube.com/video/${job.youtubeVideoId}/edit`);
  
  card.style.display = 'block';
}
```

```javascript
function auditPublishPackage(jobCard) {
  const missing = [];
  
  if (!jobCard.publishCopy?.youtube?.title)        missing.push('youtube title');
  if (!jobCard.publishCopy?.youtube?.titles?.length) missing.push('title alternatives');
  if (!jobCard.publishCopy?.youtube?.description)  missing.push('youtube description');
  if (!jobCard.publishCopy?.youtube?.tags?.length) missing.push('tags');
  if (!jobCard.publishCopy?.youtube?.pinnedComment) missing.push('pinned comment');
  if (!jobCard.thumbnailUrl)                        missing.push('thumbnail_url');
  if (!jobCard.publishCopy?.youtube?.categoryId)   missing.push('categoryId');
  if (!jobCard.publishCopy?.tiktok?.caption)       missing.push('tiktok caption');
  if (!jobCard.publishCopy?.instagram?.caption)    missing.push('instagram caption');
  
  return {
    ready: missing.length === 0,
    missing
  };
}
```

If `ready: false`, Gate 4 does not pass. Returns:
```javascript
{
  passed: false,
  outcome: 'fail_fix_in_place',
  diagnosis: [{
    severity: 'blocking',
    kind: 'publish_package_incomplete',
    issue: `Missing publish package fields: ${missing.join(', ')}`,
    fix: 'Run /generate-publish-copy and /generate-thumbnail before Gate 4'
  }]
}
```
