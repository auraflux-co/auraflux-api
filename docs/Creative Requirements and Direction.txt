Have Claude Code implement this class. It acts as the bridge between your Node server and the Python-based video engine on port 9001.
JavaScript

const axios = require('axios');

class VectCutClient {
  constructor(port = 9001) {
    this.url = `http://localhost:${port}`;
  }

  /**
   * Short-Form (9:16) Split-Screen Assembly
   * Layers the source clip on top and Bobby G on the bottom.
   */
  async assembleShortForm(asmId, clipPath, avatarPath) {
    const payload = {
      asmId,
      canvas: { width: 1080, height: 1920 },
      layers: [
        { path: clipPath,   x: 0, y: 0,   w: 1080, h: 960, z: 1 }, // Top: Clip
        { path: avatarPath, x: 0, y: 960, w: 1080, h: 960, z: 2 }  // Bottom: Bobby G
      ],
      logo: { path: 'assets/logo-80px.png', x: 985, y: 15, opacity: 0.85 }
    };
    return axios.post(`${this.url}/assemble-complex`, payload);
  }

  /**
   * Branded "Gold Ring" Overlay
   * Applies the CWN Gold border and shadow to any one-off design asset.
   */
  async burnBrandedOverlay(baseVideo, overlayImage, layout = 'TOP_RIGHT') {
    const coords = layout === 'TOP_RIGHT' ? { x: 1240, y: 40 } : { x: 40, y: 40 };
    return axios.post(`${this.url}/apply-overlay`, {
      baseVideo,
      overlayImage,
      ...coords,
      w: 640, h: 360,
      border: { size: 5, color: '#c7af4f' },
      shadow: true
    });
  }
}

module.exports = new VectCutClient();
